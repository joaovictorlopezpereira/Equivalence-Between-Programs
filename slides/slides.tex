        \vspace{0.5cm}
        \begin{itemize}
          \item $\dotprod{x}{y} = \displaystyle\sum_{i=1}^{n} x_i y_i$
        \end{itemize}
    \end{minipage}
    \pause
    \hfill
    \vrule
    \hfill
    \begin{minipage}[t]{0.50\textwidth}
        Definição do aluno 2:
        \vspace{0.5cm}
        \begin{itemize}
          \item $\dotprod{\begin{bmatrix} \hphantom{a} \end{bmatrix}}{\begin{bmatrix} \hphantom{a} \end{bmatrix}} = 0 \quad$ se $n = 0$. \pause
          \vspace{0.5cm}
          \item $\dotprod{\begin{bmatrix} x_1 \\ x_2 \end{bmatrix}}{\begin{bmatrix} y_1 \\ y_2 \end{bmatrix}} = x_1 y_1 + \dotprod{x_2}{y_2}$,

          \vspace{0.3cm}
          caso contrário, onde $x_1, y_1 \in \mathbb{R}$ e $x_2, y_2 \in \mathbb{R}^{n-1}$.
        \end{itemize}
    \end{minipage}
    \vfill
  \end{frame}

function dotprod1(x, y)
    s = 0
    for i in eachindex(x)
        s += x[i] * y[i]
    end
    return s
end
        \end{code}
    \end{minipage}
    \pause
    \hfill
    \vrule
    \hfill

function dotprod2(x, y)
    return x[1] * y[1] +
           dotprod2(x[2:end],
                    y[2:end])
end
        \end{code}
    \end{minipage}
